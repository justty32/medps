209static bool test_tick_runs_per_loaded_zone() {

212
213 // 兩個已載入的 zone，各有一個會移動的 actor

222 z2.emplace<Velocity>(e2, -1, 0);
223
224 // root 中的 actor 不應被 per-zone system tick 到

226 gm.root.emplace<Position>(er, 100, 100);
227 gm.root.emplace<Velocity>(er, 5, 5);
228
229 gm.tick();
230
231 auto& p1 = z1.get<Position>(e1);
232 auto& p2 = z2.get<Position>(e2);
233 auto& pr = gm.root.get<Position>(er);
234

236 CHECK("zone2 moved", p2.x == 9 && p2.y == 10);
237 CHECK("root untouched", pr.x == 100 && pr.y == 100);
