第 1 站 zone_key.h — 座標語言

1 個檔案，共 105 行 · 對照 CODE_TOUR.md

一個 zone 用 64-bit 的 ZoneKey 全局定址：高 16 位是 ZoneType（同時代表樹深度），其餘三段 16 位是 x / y / z。磁碟路徑由 key 推導，不另存全域清單。

World ⊃ Region ⊃ Area 嚴格三層；z 是垂直分層（地下 / 地面 / 天空），不同 z 就是不同 zone。

看什麼

  • ZoneType：type 值即樹深度，父層 type 隱含；0 保留給 ZONE_ROOT。 zone_key.h:9
  • zone_scale：尺度常數單一來源；world_dim 是唯一 per-save 執行期設定。 zone_key.h:31
  • 打包格式 ZoneType:16 | x:16 | y:16 | z:16。 zone_key.h:53
  • Area 的 x,y 是「全局 region-格」座標（world*REGION_DIM + local）。 zone_key.h:81
  • parent_of 用整數除法回推父層；z 沿鏈往上保持不變。 zone_key.h:91
  • 檔內多處 TODO: 要掛在ruleset底下 是已知的未來方向。
讀完該能回答：一個 Area key 的父 Region key 怎麼算？z 在換算鏈中如何傳遞？（答案都在 parent_of。）

4// ---- 地圖分層 (ZoneType) -------------------------------------------------
5// 嚴格階層：World ⊃ Region ⊃ Area。type 同時也代表樹的深度，所以一個 key 的
6// 父層 TYPE 是隱含的。0 保留給 ZONE_ROOT（非地圖的全局層，存放陣營 / 神祇 /
7// 具名角色）；真正的 type 從 1 開始。
8// TODO: 要掛在ruleset底下
9enum class ZoneType : uint16_t {
10 Invalid = 0, // == ZONE_ROOT
11 World = 1, // 世界層 類 Civ，約 km/格，1 回合 ≈ 1 天
12 Region = 2, // 戰略層 大戰略 / Fire-Emblem，數十 m/格
13 Area = 3, // 區域層 Rimworld / ToME4 / JRPG，約 m/格
14};
15
16// ---- 垂直分層 (z 欄位) ---------------------------------------------------
17// 三種地圖 type 都疊上同樣的三個垂直層；z 以地面為中心（往下為負，往上為正）。
18// 不同的 z 就是不同的 zone。
19// TODO: 要掛在ruleset底下
20namespace zlayer {
21 inline constexpr int16_t Underground = -1;
22 inline constexpr int16_t Ground = 0;
23 inline constexpr int16_t Sky = +1;
24}

26// ---- 尺度常數（單一來源） -------------------------------------------------
27// 全部都是估計值（「或更大」/「大約」）；座標運算必須引用這些常數，絕不要把
28// 數字寫死。world_dim 是唯一的 PER-SAVE 執行期設定（存在 ROOT 上的
29// components/world_config.h）；region/area 則是常數。
30// TODO: 要掛在ruleset底下，各種dim應該要分xyz，而不是dim^2, dim^3
31namespace zone_scale {
32 inline constexpr int16_t WORLD_DIM_DEFAULT = 200; // 預設世界地圖邊長（world-格）；見 WorldConfig
33 // 預設垂直層數：Underground / Ground / Sky（z = −1 / 0 / 1）。日後可擴充（冥界往下、
34 // 天界往上）；z 以地面為中心，層數即 zlayer 的涵蓋範圍，見 WorldConfig::world_dim_z。
35 inline constexpr int16_t WORLD_LAYERS_DEFAULT = 3;

48// ---- ZoneKey 打包 ---------------------------------------------------------
49using ZoneKey = uint64_t;
50constexpr ZoneKey ZONE_ROOT = 0; // = make_zone_key(ZoneType::Invalid, 0, 0, 0)
51
52// 位元 63-48: ZoneType (16) | 47-32: x (16) | 31-16: y (16) | 15-0: z (16)
53inline ZoneKey make_zone_key(ZoneType type, int16_t x, int16_t y, int16_t z) {
54 return (static_cast<uint64_t>(type) << 48)
55 | (static_cast<uint64_t>(static_cast<uint16_t>(x)) << 32)
56 | (static_cast<uint64_t>(static_cast<uint16_t>(y)) << 16)
57 | static_cast<uint64_t>(static_cast<uint16_t>(z));
58}
59
60inline ZoneType zone_key_type(ZoneKey k) { return static_cast<ZoneType>(k >> 48); }
61inline int16_t zone_key_x (ZoneKey k) { return static_cast<int16_t>(k >> 32); }
62inline int16_t zone_key_y (ZoneKey k) { return static_cast<int16_t>(k >> 16); }
63inline int16_t zone_key_z (ZoneKey k) { return static_cast<int16_t>(k); }
64
65// ---- 座標換算 / 階層 ------------------------------------------------------
66// 一個 zone 的 (x,y) = 它在父層 GLOBAL 格網中的座標：
67// World : 每個 z 一張地圖，位於 (0,0)。
68// Region: 它所展開的那個 world-格， x,y ∈ [0, world_dim)。
69// Area : 它所展開的那個 GLOBAL region-格，x,y ∈ [0, world_dim*REGION_DIM)。
70// 父層用整數除法回推；z 沿著鏈往上保持不變。
71
72inline ZoneKey world_key(int16_t z = zlayer::Ground) {
73 return make_zone_key(ZoneType::World, 0, 0, z);
74}
75
76inline ZoneKey region_key(int16_t world_x, int16_t world_y, int16_t z = zlayer::Ground) {
77 return make_zone_key(ZoneType::Region, world_x, world_y, z);
78}
79
80// 由父層 world-格 + region-local 偏移量建構出一個 Area key。
81inline ZoneKey area_key(int16_t world_x, int16_t world_y,
82 int16_t region_local_x, int16_t region_local_y,
83 int16_t z = zlayer::Ground) {
84 return make_zone_key(ZoneType::Area,
85 static_cast<int16_t>(world_x * zone_scale::REGION_DIM + region_local_x),
86 static_cast<int16_t>(world_y * zone_scale::REGION_DIM + region_local_y),
87 z);
88}
89
90// `k` 的直接父 zone（World 與 root 本身都回傳 ZONE_ROOT）。
91inline ZoneKey parent_of(ZoneKey k) {
92 if (k == ZONE_ROOT) return ZONE_ROOT;
93 const int16_t z = zone_key_z(k);
94 switch (zone_key_type(k)) {
95 case ZoneType::Area:
96 return make_zone_key(ZoneType::Region,
97 static_cast<int16_t>(zone_key_x(k) / zone_scale::REGION_DIM),
98 static_cast<int16_t>(zone_key_y(k) / zone_scale::REGION_DIM), z);
99 case ZoneType::Region:
100 return make_zone_key(ZoneType::World, 0, 0, z);
101 case ZoneType::World:
102 default:
103 return ZONE_ROOT;
104 }
