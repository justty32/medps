1#include "world_gen.h"
2#include <stdexcept>
3#include <string>
4#include <libtcod/mersenne.hpp>
5#include <libtcod/noise.hpp>
6
7namespace {
8
9// 一張 0~1 的 2D FBM 場。RNG 用 Mersenne Twister 鎖 seed，取樣結果決定性。
10struct Field {
11 TCODRandom rng;
12 TCODNoise noise;
13 float scale;
14 float octaves;
15
16 Field(uint32_t seed, float scale, int32_t octaves)
17 : rng{seed, TCOD_RNG_MT},
18 noise{2, TCOD_NOISE_DEFAULT_HURST, TCOD_NOISE_DEFAULT_LACUNARITY, &rng, TCOD_NOISE_SIMPLEX},
19 scale{scale},
20 octaves{static_cast<float>(octaves)} {}
21
22 float at(int x, int y) {
23 const float f[2] = {x * scale, y * scale};
24 return (noise.getFbm(f, octaves) + 1.0f) * 0.5f; // -1~1 → 0~1
25 }
26};
27
28} // namespace
29
30void world_gen::generate(const WorldGenParams& gen, tdarray<Tile>& grid, uint64_t zone_id) {
31 if (gen.width <= 0 || gen.height <= 0)
32 throw std::runtime_error(
33 "world_gen::generate: 尺寸非正: " + std::to_string(gen.width) + "x" +
34 std::to_string(gen.height) + " (zone id=" + std::to_string(zone_id) + ")");
35
36 // 高度/溫度/濕度三張獨立場，seed 派生錯開。
37 Field height{gen.seed, gen.noise_scale, gen.octaves};
38 Field temp {gen.seed + 1, gen.noise_scale, gen.octaves};
39 Field humid {gen.seed + 2, gen.noise_scale, gen.octaves};
40
41 grid.alloc(static_cast<uint32_t>(gen.width), static_cast<uint32_t>(gen.height));
42
43 grid.eachxy([&](Tile& t, int x, int y) {
44 if (height.at(x, y) <= gen.sea_level) {
45 t = Tile{TERRAIN_OCEAN, 0};
46 return;
47 }
48 uint32_t biome = TERRAIN_GRASSLAND;
49 if (temp.at(x, y) < 0.3f) biome = TERRAIN_TUNDRA;
50 else if (humid.at(x, y) >= 0.4f) biome = TERRAIN_FOREST;
51 t = Tile{biome, TILE_WALKABLE};
52 });
53}
